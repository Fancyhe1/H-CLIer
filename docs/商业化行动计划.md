#### 1.3.3 核心接口实现

**POST /api/v1/validate**

```javascript
// src/routes/validate.js
const express = require('express')
const router = express.Router()

router.post('/validate', async (req, res) => {
  const { invitation_code, machine_id, app_version } = req.body

  // 1. 参数校验
  if (!invitation_code || !machine_id) {
    return res.json({ valid: false, error_message: '缺少必要参数' })
  }

  // 2. 查询邀请码
  const code = await db.get(
    'SELECT * FROM invitation_codes WHERE code = ? AND is_active = 1',
    [invitation_code]
  )

  if (!code) {
    return res.json({ valid: false, error_message: '邀请码无效' })
  }

  // 3. 检查邀请码是否过期
  if (code.expires_at && new Date(code.expires_at) < new Date()) {
    return res.json({ valid: false, error_message: '邀请码已过期' })
  }

  // 4. 检查机器绑定数
  const activationCount = await db.get(
    'SELECT COUNT(*) as count FROM activations WHERE code = ?',
    [invitation_code]
  )

  const existingActivation = await db.get(
    'SELECT * FROM activations WHERE code = ? AND machine_id = ?',
    [invitation_code, machine_id]
  )

  if (!existingActivation && activationCount.count >= code.max_machines) {
    return res.json({
      valid: false,
      error_message: `已达到最大设备数（${code.max_machines}台）`
    })
  }

  // 5. 记录/更新激活
  if (existingActivation) {
    await db.run(
      'UPDATE activations SET last_seen_at = CURRENT_TIMESTAMP, app_version = ? WHERE code = ? AND machine_id = ?',
      [app_version, invitation_code, machine_id]
    )
  } else {
    await db.run(
      'INSERT INTO activations (code, machine_id, app_version, ip_address) VALUES (?, ?, ?, ?)',
      [invitation_code, machine_id, app_version, req.ip]
    )
  }

  // 6. 生成签名并返回
  const signature = generateSignature(invitation_code, machine_id, code.tier)

  res.json({
    valid: true,
    tier: code.tier,
    expires_at: code.expires_at || null,
    offline_grace_days: 7,
    signature
  })
})

module.exports = router
```

#### 1.3.4 邀请码生成工具

```javascript
// src/utils/code-gen.js
const crypto = require('crypto')

function generateCode() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789' // 排除易混淆字符
  const segments = []

  for (let i = 0; i < 4; i++) {
    let segment = ''
    for (let j = 0; j < 4; j++) {
      segment += chars[crypto.randomInt(chars.length)]
    }
    segments.push(segment)
  }

  return segments.join('-')
}

// 批量生成
function generateBatch(count, tier = 'pro', expiresAt = null) {
  const codes = []
  for (let i = 0; i < count; i++) {
    codes.push({
      code: generateCode(),
      tier,
      expires_at: expiresAt,
      max_machines: 1
    })
  }
  return codes
}

module.exports = { generateCode, generateBatch }
```

#### 2.2.1 在 LicenseManager 中添加功能检查方法

在 `aicoder/src-tauri/src/license.rs` 中添加：

```rust
impl LicenseManager {
    /// 检查是否为 Pro 版
    pub fn is_pro(&self) -> bool {
        let status = self.get_status().unwrap_or_default();
        status.is_activated && status.tier.as_deref() == Some("pro")
    }

    /// 检查免费版限制
    pub fn check_limit(&self, feature: &str, current_count: usize) -> Result<(), String> {
        if self.is_pro() {
            return Ok(());
        }

        match feature {
            "project_sessions" => {
                if current_count >= 3 {
                    return Err("免费版最多支持 3 个项目会话，升级 Pro 解锁无限".to_string());
                }
            }
            "terminal_sessions" => {
                if current_count >= 2 {
                    return Err("免费版最多支持 2 个终端会话，升级 Pro 解锁无限".to_string());
                }
            }
            "checkpoints_per_project" => {
                if current_count >= 3 {
                    return Err("免费版每项目最多 3 个快照，升级 Pro 解锁无限".to_string());
                }
            }
            _ => {}
        }

        Ok(())
    }
}
```

#### 2.2.2 在 lib.rs 的关键命令中添加限制检查

修改 `create_session` 命令：

```rust
#[tauri::command]
fn create_session(
    state: tauri::State<AppState>,
    project_path: String,
    title: Option<String>,
    session_type: Option<String>,
) -> Result<Session, String> {
    // 检查 license 限制
    let license = state.license_manager.lock().map_err(|e| e.to_string())?;
    let sessions = state.session_manager.lock().map_err(|e| e.to_string())?;

    let session_type_str = session_type.as_deref().unwrap_or("claude");
    let existing_count = if session_type_str == "claude" {
        sessions.get_sessions_by_type("claude").map(|s| s.len()).unwrap_or(0)
    } else {
        sessions.get_sessions_by_type("terminal").map(|s| s.len()).unwrap_or(0)
    };

    let feature = if session_type_str == "claude" {
        "project_sessions"
    } else {
        "terminal_sessions"
    };

    license.check_limit(feature, existing_count)?;
    drop(license);

    // 正常创建会话
    sessions.create_session(&project_path, title.as_deref(), session_type.as_deref())
        .map_err(|e| e.to_string())
}
```

## 合法合规清单

### 必须完成

- [ ] **用户协议**：明确服务条款、免责声明
- [ ] **隐私政策**：说明收集哪些数据（MachineGuid、IP）、用途、存储方式
- [ ] **退款政策**：建议 7 天无理由退款
- [ ] **软件著作权登记**：保护知识产权（中国版权保护中心）

### 需要注意

- **个人信息保护**：MachineGuid 属于个人信息，需在隐私政策中说明
- **支付合规**：使用正规支付渠道，保留交易记录
- **税务申报**：个人收入需申报纳税
- **开源协议**：如果部分代码开源，选择合适的协议（推荐 MIT）

### 推荐操作

1. 在应用中添加 "用户协议" 和 "隐私政策" 的同意弹窗
2. 在设置页面添加 "隐私政策" 链接
3. 每年进行一次安全审计
